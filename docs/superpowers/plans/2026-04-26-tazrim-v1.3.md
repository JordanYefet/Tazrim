```tsx
'use client'

import { useRouter, useSearchParams, usePathname } from 'next/navigation'
import { Moon, Sun } from 'lucide-react'
import { useTheme } from 'next-themes'

const MONTHS = ['ינואר','פברואר','מרץ','אפריל','מאי','יוני','יולי','אוגוסט','ספטמבר','אוקטובר','נובמבר','דצמבר']

export function Header() {
  const router = useRouter()
  const pathname = usePathname()
  const params = useSearchParams()
  const { theme, setTheme } = useTheme()

  const now = new Date()
  const year = Number(params.get('year') ?? now.getFullYear())
  const month = params.get('month') ?? ''

  function update(key: string, value: string) {
    const sp = new URLSearchParams(params.toString())
    if (value === '') {
      sp.delete(key)
    } else {
      sp.set(key, value)
    }
    router.push(`${pathname}?${sp.toString()}`)
  }

  const years = Array.from({ length: 5 }, (_, i) => now.getFullYear() - i)

  return (
    <header className="flex items-center justify-between px-4 py-3 border-b bg-card gap-4">
      <div className="flex items-center gap-2">
        <select
          className="text-sm border rounded-md px-2 py-1 bg-background"
          value={month}
          onChange={e => update('month', e.target.value)}
        >
          <option value="">הכל</option>
          {MONTHS.map((m, i) => (
            <option key={i + 1} value={i + 1}>{m}</option>
          ))}
        </select>
        <select
          className="text-sm border rounded-md px-2 py-1 bg-background"
          value={year}
          onChange={e => update('year', e.target.value)}
        >
          {years.map(y => (
            <option key={y} value={y}>{y}</option>
          ))}
        </select>
      </div>
      <button
        onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
        className="p-2 rounded-md hover:bg-accent transition-colors"
        aria-label="החלף ערכת נושא"
      >
        {theme === 'dark' ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
      </button>
    </header>
  )
}
```

```typescript
export async function getBankTransactions(params: {
  month?: number
  year?: number
  search?: string
  merchant?: string
  take?: number
  skip?: number
}) {
  const where: Prisma.BankTransactionWhereInput = {}

  if (params.year) {
    const m = params.month
    where.date = {
      gte: new Date(params.year, m ? m - 1 : 0, 1),
      lte: m
        ? new Date(params.year, m, 0, 23, 59, 59)
        : new Date(params.year, 12, 0, 23, 59, 59),
    }
  }

  if (params.merchant) {
    where.description = params.merchant
  } else if (params.search) {
    where.description = { contains: params.search }
  }

  return prisma.bankTransaction.findMany({
    where,
    orderBy: { date: 'desc' },
    ...(params.take !== undefined ? { take: params.take } : {}),
    ...(params.skip !== undefined ? { skip: params.skip } : {}),
  })
}
```

A table showing per-category totals, placed next to the donut chart. Clicking a row filters by category (same as clicking a pie slice).

- [ ] **Step 1: Create CategoryExpensesTable component**

Create `components/cards/max/category-expenses-table.tsx`:

```tsx
'use client'

import { useRouter, useSearchParams } from 'next/navigation'
import { Suspense } from 'react'
import { formatCurrency } from '@/lib/utils/currency'

interface Props {
  data: Record<string, number>
  selectedCategory?: string
}

function CategoryExpensesTableInner({ data, selectedCategory }: Props) {
  const router = useRouter()
  const params = useSearchParams()

  const rows = Object.entries(data)
    .map(([name, total]) => ({ name, total }))
    .sort((a, b) => b.total - a.total)

  function handleClick(name: string) {
    const sp = new URLSearchParams(params.toString())
    if (selectedCategory === name) {
      sp.delete('category')
    } else {
      sp.set('category', name)
    }
    router.push(`/cards/max?${sp.toString()}`)
  }

  return (
    <div className="rounded-lg border bg-card p-4">
      <h3 className="font-semibold mb-4">פירוט לפי קטגוריה</h3>
      <div className="overflow-y-auto max-h-72 md:max-h-80">
        <table className="w-full text-sm">
          <thead>
            <tr className="border-b text-muted-foreground">
              <th className="text-right py-1.5 px-2 font-medium">קטגוריה</th>
              <th className="text-right py-1.5 px-2 font-medium">סכום</th>
            </tr>
          </thead>
          <tbody>
            {rows.map(row => (
              <tr
                key={row.name}
                className="border-b hover:bg-accent/30 cursor-pointer transition-colors"
                style={{ opacity: selectedCategory && selectedCategory !== row.name ? 0.5 : 1 }}
                onClick={() => handleClick(row.name)}
              >
                <td className="py-1.5 px-2">{row.name}</td>
                <td className="py-1.5 px-2 text-red-600 dark:text-red-400">
                  {formatCurrency(row.total)}
                </td>
              </tr>
            ))}
          </tbody>
        </table>
        {rows.length === 0 && (
          <p className="text-center py-6 text-muted-foreground text-sm">אין נתונים</p>
        )}
      </div>
    </div>
  )
}

export function CategoryExpensesTable(props: Props) {
  return (
    <Suspense fallback={<div className="rounded-lg border bg-card p-4 h-48" />}>
      <CategoryExpensesTableInner {...props} />
    </Suspense>
  )
}
```
