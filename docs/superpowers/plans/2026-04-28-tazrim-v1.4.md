interface SearchParams {
  year?: string
  month?: string
  search?: string
  merchant?: string
  showIgnored?: string
}

async function getBankData(params: SearchParams) {
  const year = params.year ?? String(new Date().getFullYear())
  const base = 'http://localhost:3000'

  const sp = new URLSearchParams({ year })
  if (params.month) sp.set('month', params.month)
  if (params.search) sp.set('search', params.search)
  if (params.merchant) sp.set('merchant', params.merchant)
  if (params.showIgnored) sp.set('showIgnored', params.showIgnored)

  const [txRes, summaryRes] = await Promise.all([
    fetch(`${base}/api/bank/transactions?${sp}`, { cache: 'no-store' }),
    fetch(`${base}/api/summary?year=${year}`, { cache: 'no-store' }),
  ])

  const transactions = await txRes.json()
  const summary = await summaryRes.json()
  return {
    transactions,
    bankSummary: summary.bank as Record<string, { income: number; expense: number }>,
  }
}

export default async function BankPage({
  searchParams,
}: {
  searchParams: Promise<SearchParams>
}) {
  const params = await searchParams
  const year = params.year ?? String(new Date().getFullYear())
  const { transactions, bankSummary } = await getBankData(params)
  const month = params.month ? Number(params.month) : undefined

  const income = month
    ? (bankSummary[String(month)]?.income ?? 0)
    : Object.values(bankSummary).reduce((s, m) => s + m.income, 0)
  const expense = month
    ? (bankSummary[String(month)]?.expense ?? 0)
    : Object.values(bankSummary).reduce((s, m) => s + m.expense, 0)

  const averageExpense = !month
    ? expense / Math.max(Object.values(bankSummary).filter(m => m.expense > 0).length, 1)
    : undefined

  const chartData = Object.entries(bankSummary).map(([m, d]) => ({
    month: Number(m),
    income: d.income,
    expense: d.expense,
  }))

  const balancePoints = transactions.map((tx: { date: string; balance: number }) => ({
    date: tx.date,
    balance: tx.balance,
  }))

  const fetchSp = new URLSearchParams({ year })
  if (params.search) fetchSp.set('search', params.search)
  if (params.merchant) fetchSp.set('merchant', params.merchant)
  if (params.showIgnored) fetchSp.set('showIgnored', params.showIgnored)
  const hasMore = !params.month && transactions.length >= 50

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold">בנק לאומי</h1>
      <BankSummaryCards income={income} expense={expense} averageExpense={averageExpense} />

export async function upsertAliasAndBulkUpdate(params: {
  source: AliasSource
  originalName: string
  displayName: string
  categoryOverride?: string
}) {
  const alias = await prisma.transactionAlias.upsert({
    where: { source_originalName: { source: params.source, originalName: params.originalName } },
    update: { displayName: params.displayName, categoryOverride: params.categoryOverride },
    create: params,
  })

  if (params.categoryOverride && (params.source === 'MAX' || params.source === 'HITECH')) {
    await prisma.creditCardTransaction.updateMany({
      where: { originalMerchantName: params.originalName },
      data: { categoryOverride: params.categoryOverride },
    })
  }

  if (params.source === 'BANK') {
    await prisma.bankTransaction.updateMany({
      where: { originalDescription: params.originalName },
      data: { description: params.displayName },
    })
  }

  return alias
}

'use client'

import { useState } from 'react'
import { Trash2 } from 'lucide-react'

interface Alias {
  id: string
  source: string
  originalName: string
  displayName: string
  categoryOverride: string | null
}

interface Category { id: string; name: string }

const SOURCE_LABELS: Record<string, string> = { BANK: 'בנק לאומי', MAX: 'מקס', HITECH: 'הייטקזון' }

export function MerchantMappings({ initial, categories }: { initial: Alias[]; categories: Category[] }) {
  const [aliases, setAliases] = useState(initial)
  const [editingId, setEditingId] = useState<string | null>(null)
  const [editDisplay, setEditDisplay] = useState('')

  async function saveDisplayName(alias: Alias) {
    await fetch('/api/aliases', {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        source: alias.source,
        originalName: alias.originalName,
        displayName: editDisplay,
        categoryOverride: alias.categoryOverride,
      }),
    })
    setAliases(prev => prev.map(a => a.id === alias.id ? { ...a, displayName: editDisplay } : a))
    setEditingId(null)
  }

  async function saveCategory(alias: Alias, categoryOverride: string) {
    await fetch('/api/aliases', {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        source: alias.source,
        originalName: alias.originalName,
        displayName: alias.displayName,
        categoryOverride,
      }),
    })
    setAliases(prev => prev.map(a => a.id === alias.id ? { ...a, categoryOverride } : a))
  }

  async function deleteAlias(alias: Alias) {
    if (!window.confirm(`למחוק את המיפוי של "${alias.originalName}"?`)) return
    await fetch(`/api/aliases/${alias.id}`, { method: 'DELETE' })
    setAliases(prev => prev.filter(a => a.id !== alias.id))
  }

  if (!aliases.length) {
    return <p className="text-sm text-muted-foreground">לא קיימים מיפויים עדיין. העלה קבצים כדי להתחיל.</p>
  }

  return (
    <div className="overflow-x-auto">
      <table className="w-full text-sm">
        <thead>
          <tr className="border-b text-muted-foreground">
            <th className="text-right py-2 px-3 font-medium">שם מקורי</th>
            <th className="text-right py-2 px-3 font-medium">שם תצוגה</th>
            <th className="text-right py-2 px-3 font-medium">קטגוריה</th>
            <th className="text-right py-2 px-3 font-medium">מקור</th>
            <th className="py-2 px-3"></th>
          </tr>
        </thead>
        <tbody>
          {aliases.map(alias => (
            <tr key={alias.id} className="border-b hover:bg-accent/30">
              <td className="py-2 px-3 text-muted-foreground">{alias.originalName}</td>
              <td className="py-2 px-3">
                {editingId === alias.id ? (
                  <input
                    autoFocus
                    className="border rounded px-1 py-0.5 text-sm bg-background w-full"
                    value={editDisplay}
                    onChange={e => setEditDisplay(e.target.value)}
                    onBlur={() => saveDisplayName(alias)}
                    onKeyDown={e => { if (e.key === 'Enter') saveDisplayName(alias); if (e.key === 'Escape') setEditingId(null) }}
                  />
                ) : (
                  <span
                    className="cursor-pointer hover:text-primary"
                    onClick={() => { setEditingId(alias.id); setEditDisplay(alias.displayName) }}
                  >
                    {alias.displayName}
                  </span>
                )}
              </td>
              <td className="py-2 px-3">
                <select
                  className="text-sm border rounded px-1 py-0.5 bg-background"
                  value={alias.categoryOverride ?? ''}
                  onChange={e => saveCategory(alias, e.target.value)}
                >
                  <option value="">— ללא —</option>
                  {categories.map(c => (
                    <option key={c.id} value={c.name}>{c.name}</option>
                  ))}
                </select>
              </td>
              <td className="py-2 px-3">{SOURCE_LABELS[alias.source] ?? alias.source}</td>
              <td className="py-2 px-3 text-center">
                <button
                  onClick={() => deleteAlias(alias)}
                  aria-label="מחק מיפוי"
                  className="p-0.5 rounded hover:bg-accent transition-colors"
                >
                  <Trash2 className="h-4 w-4 text-muted-foreground hover:text-destructive" />
                </button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

export async function upsertCCTransactions(
  transactions: ParsedCCTransaction[],
  fileName: string
): Promise<{ created: number; skipped: number }> {
  const provider = transactions[0]?.cardProvider ?? 'MAX'
  const source = provider === 'MAX' ? 'MAX' : 'HITECH'

  const aliases = await prisma.transactionAlias.findMany({ where: { source } })
  const aliasMap = new Map(aliases.map(a => [a.originalName, a.displayName]))

  let created = 0
  let skipped = 0

  for (const tx of transactions) {
    const contentHash = [
      tx.transactionDate.toISOString().slice(0, 10),
      tx.originalMerchantName,
      tx.chargedAmount,
      tx.cardProvider,
    ].join('|')

    const crossFileDuplicate = await prisma.creditCardTransaction.findFirst({
      where: { contentHash, sourceFile: { not: fileName } },
    })
    if (crossFileDuplicate) { skipped++; continue }

    await prisma.creditCardTransaction.create({
      data: {
        transactionDate: tx.transactionDate,
        billingDate: tx.billingDate,
        originalMerchantName: tx.originalMerchantName,
        merchantName: aliasMap.get(tx.originalMerchantName) ?? tx.originalMerchantName,
        category: tx.category,
        chargedAmount: tx.chargedAmount,
        originalAmount: tx.originalAmount,
        originalCurrency: tx.originalCurrency,
        cardLastFour: tx.cardLastFour,
        transactionType: tx.transactionType,
        notes: tx.notes,
        cardProvider: tx.cardProvider as CardProvider,
        contentHash,
        sourceFile: fileName,
      },
    })
    created++
  }

  return { created, skipped }
}

import { NextRequest, NextResponse } from 'next/server'
import { parseBankLeumi } from '@/lib/parsers/bank-leumi'
import { parseMaxCC } from '@/lib/parsers/max-cc'
import { upsertBankTransactions } from '@/lib/db/bank'
import { upsertCCTransactions } from '@/lib/db/cards'
import { recordUpload } from '@/lib/db/upload'
import { prisma } from '@/lib/db/client'

export async function POST(req: NextRequest) {
  const form = await req.formData()
  const file = form.get('file') as File | null
  const source = form.get('source') as string | null

  if (!file || !source) {
    return NextResponse.json({ error: 'חסר קובץ או סוג מקור' }, { status: 400 })
  }

  const buffer = Buffer.from(await file.arrayBuffer())

  if (source === 'BANK') {
    const { transactions, errors } = parseBankLeumi(buffer, file.name)
    if (errors.length) return NextResponse.json({ error: errors[0] }, { status: 422 })

    const result = await upsertBankTransactions(transactions)
    await recordUpload({ fileName: file.name, recordCount: result.created, source: 'BANK' })
    return NextResponse.json(result)
  }

  if (source === 'MAX') {
    const { transactions, errors } = parseMaxCC(buffer)
    if (errors.length) return NextResponse.json({ error: errors[0] }, { status: 422 })

    const existingUpload = await prisma.uploadHistory.findFirst({
      where: { fileName: file.name, source: 'MAX' },
    })
    if (existingUpload) {
      return NextResponse.json(
        { error: 'קובץ זה כבר הועלה בעבר', duplicate: true },
        { status: 409 }
      )
    }

    const result = await upsertCCTransactions(transactions, file.name)
    await recordUpload({ fileName: file.name, recordCount: result.created, source: 'MAX' })

    const txCategories = [...new Set(transactions.map(t => t.category).filter(Boolean))]
    const existing = await prisma.category.findMany({ where: { name: { in: txCategories } } })
    const existingNames = new Set(existing.map(c => c.name))
    const newCategories = txCategories.filter(name => !existingNames.has(name))

    return NextResponse.json({ ...result, newCategories })
  }

  return NextResponse.json({ error: 'מקור לא מוכר' }, { status: 400 })
}
