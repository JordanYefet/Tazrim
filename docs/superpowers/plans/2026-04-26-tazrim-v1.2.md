```typescript
export async function getCCTransactions(params: {
  provider: CardProvider
  month?: number
  year?: number
  search?: string
  merchant?: string
  category?: string
  showIgnored?: boolean
}) {
  const where: Prisma.CreditCardTransactionWhereInput = {
    cardProvider: params.provider,
  }

  if (!params.showIgnored) {
    where.ignored = false
  }

  if (params.year) {
    const m = params.month
    where.transactionDate = {
      gte: new Date(params.year, m ? m - 1 : 0, 1),
      lte: m
        ? new Date(params.year, m, 0, 23, 59, 59)
        : new Date(params.year, 12, 0, 23, 59, 59),
    }
  }

  if (params.merchant) {
    where.merchantName = params.merchant
  } else if (params.search) {
    where.merchantName = { contains: params.search }
  }

  if (params.category) {
    where.OR = [
      { categoryOverride: params.category },
      { category: params.category, categoryOverride: null },
    ]
  }

  return prisma.creditCardTransaction.findMany({
    where,
    orderBy: { transactionDate: 'desc' },
  })
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { parseBankLeumi } from '@/lib/parsers/bank-leumi'
import { parseMaxCC } from '@/lib/parsers/max-cc'
import { upsertBankTransactions } from '@/lib/db/bank'
import { upsertCCTransactions } from '@/lib/db/cards'
import { recordUpload } from '@/lib/db/upload'
import { prisma } from '@/lib/db/client'

export async function POST(req: NextRequest) {
  const form = await req.formData()
  const file = form.get('file') as File | null
  const source = form.get('source') as string | null

  if (!file || !source) {
    return NextResponse.json({ error: 'חסר קובץ או סוג מקור' }, { status: 400 })
  }

  const buffer = Buffer.from(await file.arrayBuffer())

  if (source === 'BANK') {
    const { transactions, errors } = parseBankLeumi(buffer, file.name)
    if (errors.length) return NextResponse.json({ error: errors[0] }, { status: 422 })

    const result = await upsertBankTransactions(transactions)
    await recordUpload({ fileName: file.name, recordCount: result.created, source: 'BANK' })
    return NextResponse.json(result)
  }

  if (source === 'MAX') {
    const { transactions, errors } = parseMaxCC(buffer)
    if (errors.length) return NextResponse.json({ error: errors[0] }, { status: 422 })

    const result = await upsertCCTransactions(transactions)
    await recordUpload({ fileName: file.name, recordCount: result.created, source: 'MAX' })

    const txCategories = [...new Set(transactions.map(t => t.category).filter(Boolean))]
    const existing = await prisma.category.findMany({ where: { name: { in: txCategories } } })
    const existingNames = new Set(existing.map(c => c.name))
    const newCategories = txCategories.filter(name => !existingNames.has(name))

    return NextResponse.json({ ...result, newCategories })
  }

  return NextResponse.json({ error: 'מקור לא מוכר' }, { status: 400 })
}
```

```typescript
'use client'

import { PieChart, Pie, Cell, Tooltip, Legend, ResponsiveContainer } from 'recharts'
import { useRouter, useSearchParams } from 'next/navigation'
import { Suspense } from 'react'
import { formatCurrency } from '@/lib/utils/currency'

const COLORS = ['#22c55e','#3b82f6','#f59e0b','#ef4444','#8b5cf6','#06b6d4','#f97316','#ec4899','#84cc16','#14b8a6']

function CategoryDonutInner({
  data,
  selectedCategory,
}: {
  data: Record<string, number>
  selectedCategory?: string
}) {
  const router = useRouter()
  const params = useSearchParams()

  const chartData = Object.entries(data)
    .map(([name, value]) => ({ name, value }))
    .sort((a, b) => b.value - a.value)

  function handleClick(entry: { name: string }) {
    const sp = new URLSearchParams(params.toString())
    if (selectedCategory === entry.name) {
      sp.delete('category')
    } else {
      sp.set('category', entry.name)
    }
    router.push(`/cards/max?${sp.toString()}`)
  }

  return (
    <div className="rounded-lg border bg-card p-4">
      <h3 className="font-semibold mb-4">הוצאות לפי קטגוריה</h3>
      <div className="h-48 sm:h-64 md:h-72">
        <ResponsiveContainer width="100%" height="100%">
          <PieChart>
            <Pie
              data={chartData}
              dataKey="value"
              nameKey="name"
              innerRadius="35%"
              outerRadius="60%"
              paddingAngle={2}
              onClick={handleClick}
              style={{ cursor: 'pointer' }}
            >
              {chartData.map((entry, i) => (
                <Cell
                  key={i}
                  fill={COLORS[i % COLORS.length]}
                  opacity={selectedCategory && selectedCategory !== entry.name ? 0.35 : 1}
                />
              ))}
            </Pie>
            <Tooltip formatter={(v: number) => formatCurrency(v)} />
            <Legend />
          </PieChart>
        </ResponsiveContainer>
      </div>
    </div>
  )
}

export function CategoryDonut({
  data,
  selectedCategory,
}: {
  data: Record<string, number>
  selectedCategory?: string
}) {
  return (
    <Suspense fallback={<div className="rounded-lg border bg-card p-4 h-48 sm:h-64 md:h-72" />}>
      <CategoryDonutInner data={data} selectedCategory={selectedCategory} />
    </Suspense>
  )
}
```

```typescript
'use client'

import { useState } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import { formatDate } from '@/lib/utils/date'
import { formatCurrency } from '@/lib/utils/currency'
import { X, EyeOff, Eye } from 'lucide-react'

interface CCTx {
  id: string
  transactionDate: string
  billingDate: string
  merchantName: string
  originalMerchantName: string
  category: string
  categoryOverride: string | null
  chargedAmount: number
  originalAmount: number
  originalCurrency: string
  ignored: boolean
}

interface Category { id: string; name: string }

export function MaxTransactionsTable({
  transactions: initial,
  categories,
  activeCategory,
  showIgnored,
}: {
  transactions: CCTx[]
  categories: Category[]
  activeCategory?: string
  showIgnored?: boolean
}) {
  const router = useRouter()
  const params = useSearchParams()
  const [transactions, setTransactions] = useState(initial)
  const activeMerchant = params.get('merchant')

  async function saveCategory(tx: CCTx, categoryOverride: string) {
    await fetch(`/api/cards/max/transactions/${tx.id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ categoryOverride }),
    })
    setTransactions(prev =>
      prev.map(t =>
        t.originalMerchantName === tx.originalMerchantName ? { ...t, categoryOverride } : t
      )
    )
  }

  async function toggleIgnored(tx: CCTx) {
    const ignored = !tx.ignored
    await fetch(`/api/cards/max/transactions/${tx.id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ignored }),
    })
    if (showIgnored) {
      setTransactions(prev => prev.map(t => t.id === tx.id ? { ...t, ignored } : t))
    } else {
      setTransactions(prev => prev.filter(t => t.id !== tx.id))
    }
  }

  function filterMerchant(name: string) {
    const sp = new URLSearchParams(params.toString())
    sp.set('merchant', name)
    router.push(`/cards/max?${sp.toString()}`)
  }

  function clearMerchant() {
    const sp = new URLSearchParams(params.toString())
    sp.delete('merchant')
    router.push(`/cards/max?${sp.toString()}`)
  }

  function clearCategory() {
    const sp = new URLSearchParams(params.toString())
    sp.delete('category')
    router.push(`/cards/max?${sp.toString()}`)
  }

  function toggleShowIgnored() {
    const sp = new URLSearchParams(params.toString())
    if (showIgnored) {
      sp.delete('showIgnored')
    } else {
      sp.set('showIgnored', 'true')
    }
    router.push(`/cards/max?${sp.toString()}`)
  }

  const hasFilter = activeMerchant || activeCategory

  return (
    <div className="space-y-2">
      <div className="flex items-center justify-between flex-wrap gap-2">
        {hasFilter ? (
          <div className="flex items-center gap-2 text-sm flex-wrap">
            <span className="text-muted-foreground">מסנן:</span>
            {activeMerchant && (
              <span className="inline-flex items-center gap-1 bg-primary/10 text-primary px-2 py-0.5 rounded-full">
                {activeMerchant}
                <button onClick={clearMerchant} aria-label="הסר מסנן בית עסק">
                  <X className="h-3 w-3" />
                </button>
              </span>
            )}
            {activeCategory && (
              <span className="inline-flex items-center gap-1 bg-primary/10 text-primary px-2 py-0.5 rounded-full">
                {activeCategory}
                <button onClick={clearCategory} aria-label="הסר מסנן קטגוריה">
                  <X className="h-3 w-3" />
                </button>
              </span>
            )}
          </div>
        ) : <div />}
        <label className="flex items-center gap-2 text-sm text-muted-foreground cursor-pointer select-none">
          <input
            type="checkbox"
            checked={!!showIgnored}
            onChange={toggleShowIgnored}
            className="accent-primary"
          />
          הצג מוסתרים
        </label>
      </div>

      <div className="rounded-lg border bg-card overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-sm">
            <thead>
              <tr className="border-b bg-muted/40">
                {['תאריך עסקה', 'שם בית עסק', 'קטגוריה', 'סכום חיוב', 'תאריך חיוב', ''].map((h, i) => (
                  <th key={i} className="text-right py-2 px-3 font-medium text-muted-foreground">{h}</th>
                ))}
              </tr>
            </thead>
            <tbody>
              {transactions.map(tx => (
                <tr
                  key={tx.id}
                  className="border-b hover:bg-accent/30 transition-colors"
                  style={tx.ignored ? { opacity: 0.4 } : undefined}
                >
                  <td className="py-2 px-3 whitespace-nowrap">{formatDate(tx.transactionDate)}</td>
                  <td className="py-2 px-3 min-w-[160px]">
                    {tx.ignored ? (
                      <span className="line-through text-muted-foreground">{tx.merchantName}</span>
                    ) : (
                      <span
                        className="cursor-pointer hover:underline hover:text-primary"
                        onClick={() => filterMerchant(tx.merchantName)}
                      >
                        {tx.merchantName}
                      </span>
                    )}
                  </td>
                  <td className="py-2 px-3 min-w-[140px]">
                    {tx.ignored ? (
                      <span className="line-through text-muted-foreground">
                        {tx.categoryOverride ?? tx.category}
                      </span>
                    ) : (
                      <select
                        className="text-sm border rounded px-1 py-0.5 bg-background w-full"
                        value={tx.categoryOverride ?? tx.category}
                        onChange={e => saveCategory(tx, e.target.value)}
                      >
                        <option value={tx.category}>{tx.category}</option>
                        {categories
                          .filter(c => c.name !== tx.category)
                          .map(c => (
                            <option key={c.id} value={c.name}>{c.name}</option>
                          ))}
                      </select>
                    )}
                  </td>
                  <td
                    className="py-2 px-3 whitespace-nowrap"
                    style={
                      tx.ignored
                        ? { textDecoration: 'line-through', color: 'var(--muted-foreground)' }
                        : { color: 'rgb(220 38 38)' }
                    }
                  >
                    {formatCurrency(tx.chargedAmount)}
                    {!tx.ignored && tx.originalCurrency !== '₪' && (
                      <span className="text-xs text-muted-foreground mr-1">
                        ({tx.originalAmount} {tx.originalCurrency})
                      </span>
                    )}
                  </td>
                  <td className="py-2 px-3 whitespace-nowrap">{formatDate(tx.billingDate)}</td>
                  <td className="py-2 px-3 text-center">
                    <button
                      onClick={() => toggleIgnored(tx)}
                      aria-label={tx.ignored ? 'בטל התעלמות' : 'התעלם מעסקה'}
                      className="p-0.5 rounded hover:bg-accent transition-colors"
                    >
                      {tx.ignored
                        ? <Eye className="h-4 w-4 text-primary" />
                        : <EyeOff className="h-4 w-4 text-muted-foreground" />
                      }
                    </button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
          {transactions.length === 0 && (
            <p className="text-center py-8 text-muted-foreground text-sm">אין עסקאות לתקופה זו</p>
          )}
        </div>
      </div>
    </div>
  )
}
```

```typescript
import { MaxSummaryCards } from '@/components/cards/max/max-summary-cards'
import { CategoryDonut } from '@/components/cards/max/category-donut'
import { MonthlySpendingChart } from '@/components/cards/max/monthly-spending-chart'
import { MaxTransactionsTable } from '@/components/cards/max/max-transactions-table'

interface SearchParams {
  year?: string
  month?: string
  search?: string
  merchant?: string
  category?: string
  showIgnored?: string
}

async function getMaxData(params: SearchParams) {
  const year = params.year ?? String(new Date().getFullYear())
  const month = params.month ?? String(new Date().getMonth() + 1)
  const base = 'http://localhost:3000'

  const sp = new URLSearchParams({ year, month })
  if (params.search) sp.set('search', params.search)
  if (params.merchant) sp.set('merchant', params.merchant)
  if (params.category) sp.set('category', params.category)
  if (params.showIgnored) sp.set('showIgnored', params.showIgnored)

  const [txRes, summaryRes, catRes, categoriesRes] = await Promise.all([
    fetch(`${base}/api/cards/max/transactions?${sp}`, { cache: 'no-store' }),
    fetch(`${base}/api/summary?year=${year}`, { cache: 'no-store' }),
    fetch(`${base}/api/cards/max/transactions?year=${year}&month=${month}`, { cache: 'no-store' }),
    fetch(`${base}/api/categories`, { cache: 'no-store' }),
  ])

  const transactions = await txRes.json()
  const summary = await summaryRes.json()
  const monthlyTxs = await catRes.json()
  const categories = await categoriesRes.json()

  const categoryTotals: Record<string, number> = {}
  for (const tx of monthlyTxs) {
    const cat = tx.categoryOverride ?? tx.category
    categoryTotals[cat] = (categoryTotals[cat] ?? 0) + tx.chargedAmount
  }

  return { transactions, maxSummary: summary.max, categoryTotals, categories }
}

export default async function MaxPage({
  searchParams,
}: {
  searchParams: Promise<SearchParams>
}) {
  const params = await searchParams
  const { transactions, maxSummary, categoryTotals, categories } = await getMaxData(params)
  const month = Number(params.month ?? new Date().getMonth() + 1)
  const total = transactions.reduce((s: number, t: { chargedAmount: number }) => s + t.chargedAmount, 0)

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold">מקס כרטיס אשראי</h1>
      <MaxSummaryCards total={total} count={transactions.length} />
      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
        <CategoryDonut data={categoryTotals} selectedCategory={params.category} />
        <MonthlySpendingChart data={maxSummary} selectedMonth={month} />
      </div>
      <MaxTransactionsTable
        key={[params.merchant, params.month, params.year, params.category, params.showIgnored].join('|')}
        transactions={transactions}
        categories={categories}
        activeCategory={params.category}
        showIgnored={params.showIgnored === 'true'}
      />
    </div>
  )
}
```

```typescript
'use client'

import { useState, useRef } from 'react'

const SOURCES = [
  { value: 'BANK', label: 'בנק לאומי' },
  { value: 'MAX', label: 'מקס כרטיס אשראי' },
]

interface UploadResult {
  created: number
  skipped: number
  newCategories?: string[]
}

export function UploadZone({ onSuccess }: { onSuccess: () => void }) {
  const [source, setSource] = useState('BANK')
  const [dragging, setDragging] = useState(false)
  const [loading, setLoading] = useState(false)
  const [result, setResult] = useState<UploadResult | null>(null)
  const [error, setError] = useState<string | null>(null)
  const [pendingCategories, setPendingCategories] = useState<string[]>([])
  const [selectedCategories, setSelectedCategories] = useState<Set<string>>(new Set())
  const [categoryLoading, setCategoryLoading] = useState(false)
  const inputRef = useRef<HTMLInputElement>(null)

  async function upload(file: File) {
    setLoading(true)
    setResult(null)
    setError(null)
    setPendingCategories([])
    setSelectedCategories(new Set())

    const form = new FormData()
    form.append('file', file)
    form.append('source', source)

    try {
      const res = await fetch('/api/upload', { method: 'POST', body: form })
      const data = await res.json()
      if (!res.ok) { setError(data.error ?? 'שגיאה בהעלאה'); return }
      setResult(data)
      onSuccess()
      if (source === 'MAX' && data.newCategories?.length) {
        setPendingCategories(data.newCategories)
        setSelectedCategories(new Set(data.newCategories))
      }
    } catch {
      setError('שגיאת רשת')
    } finally {
      setLoading(false)
    }
  }

  function toggleCategory(name: string) {
    setSelectedCategories(prev => {
      const next = new Set(prev)
      if (next.has(name)) next.delete(name)
      else next.add(name)
      return next
    })
  }

  async function addSelectedCategories() {
    setCategoryLoading(true)
    for (const name of selectedCategories) {
      await fetch('/api/categories', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name }),
      })
    }
    setCategoryLoading(false)
    setPendingCategories([])
  }

  function onDrop(e: React.DragEvent) {
    e.preventDefault()
    setDragging(false)
    const file = e.dataTransfer.files[0]
    if (file) upload(file)
  }

  return (
    <div className="space-y-4">
      <div className="flex gap-2">
        {SOURCES.map(s => (
          <button
            key={s.value}
            onClick={() => setSource(s.value)}
            className={`px-4 py-2 rounded-md text-sm border transition-colors ${
              source === s.value
                ? 'bg-primary text-primary-foreground border-primary'
                : 'bg-background hover:bg-accent border-border'
            }`}
          >
            {s.label}
          </button>
        ))}
      </div>

      <div
        onDragOver={e => { e.preventDefault(); setDragging(true) }}
        onDragLeave={() => setDragging(false)}
        onDrop={onDrop}
        onClick={() => inputRef.current?.click()}
        className={`border-2 border-dashed rounded-lg p-12 text-center cursor-pointer transition-colors ${
          dragging ? 'border-primary bg-primary/5' : 'border-border hover:border-primary/50'
        }`}
      >
        <p className="text-muted-foreground text-sm">גרור קובץ לכאן או לחץ לבחירה</p>
        <p className="text-xs text-muted-foreground mt-1">
          {source === 'BANK' ? '.xls — בנק לאומי' : '.xlsx — מקס'}
        </p>
        <input
          ref={inputRef}
          type="file"
          className="hidden"
          accept=".xls,.xlsx"
          onChange={e => { const f = e.target.files?.[0]; if (f) upload(f) }}
        />
      </div>

      {loading && <p className="text-sm text-muted-foreground">מעלה...</p>}
      {error && <p className="text-sm text-destructive">{error}</p>}
      {result && (
        <div className="text-sm p-3 rounded-md bg-accent">
          <span className="text-primary font-medium">{result.created} עסקאות חדשות</span>
          {result.skipped > 0 && (
            <span className="text-muted-foreground"> · {result.skipped} כפולות דולגו</span>
          )}
        </div>
      )}

      {pendingCategories.length > 0 && (
        <div className="rounded-lg border bg-card p-4 space-y-3 text-sm">
          <div>
            <p className="font-semibold">נמצאו קטגוריות חדשות בקובץ</p>
            <p className="text-muted-foreground text-xs mt-0.5">בחר אילו קטגוריות להוסיף לרשימה שלך:</p>
          </div>
          <div className="flex flex-col gap-2">
            {pendingCategories.map(name => (
              <label key={name} className="flex items-center gap-2 cursor-pointer">
                <input
                  type="checkbox"
                  checked={selectedCategories.has(name)}
                  onChange={() => toggleCategory(name)}
                  className="accent-primary w-4 h-4"
                />
                <span>{name}</span>
              </label>
            ))}
          </div>
          <div className="flex gap-2">
            <button
              onClick={addSelectedCategories}
              disabled={categoryLoading || selectedCategories.size === 0}
              className="bg-primary text-primary-foreground px-3 py-1.5 rounded-md text-sm disabled:opacity-50"
            >
              {categoryLoading ? 'מוסיף...' : 'הוסף נבחרים'}
            </button>
            <button
              onClick={() => setPendingCategories([])}
              className="border px-3 py-1.5 rounded-md text-sm text-muted-foreground hover:bg-accent"
            >
              דלג
            </button>
          </div>
        </div>
      )}
    </div>
  )
}
```
